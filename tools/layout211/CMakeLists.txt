cmake_minimum_required(VERSION 3.25)
project(Layout211 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_executable(layout211 layout211.cpp)
target_compile_definitions(layout211 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _ITERATOR_DEBUG_LEVEL=0)
target_include_directories(layout211 PRIVATE "${UPSTREAM}" "${UPSTREAM}/External/imgui")
foreach(name official spectator)
    add_executable(${name}_layout "../${name}_layout_probe.cpp")
    target_compile_definitions(${name}_layout PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _ITERATOR_DEBUG_LEVEL=0)
    target_include_directories(${name}_layout PRIVATE "${UPSTREAM}" "${UPSTREAM}/External/imgui")
endforeach()

add_library(imgui_contract SHARED "${UPSTREAM}/External/imgui/imgui.cpp" "${UPSTREAM}/External/imgui/imgui_draw.cpp" "${UPSTREAM}/External/imgui/imgui_tables.cpp" "${UPSTREAM}/External/imgui/imgui_widgets.cpp" imgui_contract.cpp)
target_include_directories(imgui_contract PRIVATE "${UPSTREAM}/External/imgui")
target_link_options(imgui_contract PRIVATE /MAP)
